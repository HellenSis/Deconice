<?php
/**

// 1. config/db.php (Seguridad: Prepared Statements + PDO)
$db_config = [
    'host' => 'localhost',
    'dbname' => 'deconice_db',
    'user' => 'user_db',
    'pass' => 'password_db'
];

try {
    $pdo = new PDO("mysql:host={$db_config['host']};dbname={$db_config['dbname']};charset=utf8mb4", $db_config['user'], $db_config['pass'], [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        PDO::ATTR_EMULATE_PREPARES => false,
    ]);
} catch (PDOException $e) {
    error_log($e->getMessage());
    die("Error de conexión. Consulte al administrador.");
}

// 2. includes/security.php (CSRF, Sanitización, Headers)
session_start(['cookie_httponly' => true, 'cookie_secure' => true, 'cookie_samesite' => 'Strict']);

function generate_csrf_token() {
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function verify_csrf_token($token) {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

function clean_input($data) {
    return htmlspecialchars(trim($data), ENT_QUOTES, 'UTF-8');
}

// Security Headers
header("Content-Security-Policy: default-src 'self' https://cdn.jsdelivr.net https://wa.me; script-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net; style-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net;");
header("X-Frame-Options: DENY");
header("X-Content-Type-Options: nosniff");
header("Referrer-Policy: strict-origin-when-cross-origin");

// 3. procesar-cotizacion.php (Ejemplo de Guardado Seguro)
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!verify_csrf_token($_POST['csrf_token'])) {
        die("Error de validación de seguridad.");
    }

    $nombre = filter_var($_POST['nombre'], FILTER_SANITIZE_STRING);
    $celular = filter_var($_POST['celular'], FILTER_SANITIZE_NUMBER_INT);
    // ... más validaciones

    $stmt = $pdo->prepare("INSERT INTO cotizaciones (producto_categoria, medidas, departamento, nombre, celular, ip_origen) VALUES (?, ?, ?, ?, ?, ?)");
    $stmt->execute([$_POST['categoria'], $_POST['medidas'], $_POST['departamento'], $nombre, $celular, $_SERVER['REMOTE_ADDR']]);
    
    // Redirección segura o envío de correo vía PHPMailer
}
?>
